pipeline {
    agent any

    stages {

        stage('Checkout Source') {
            steps {
                echo 'Cloning repository from GitHub...'
                checkout scm
            }
        }

        stage('Install Dependencies') {
            steps {
                echo 'Installing Node packages/modules...'
                bat 'npm install'
            }
        }

        stage('Run Tests') {
            steps {
                echo 'Running local test script...'
                bat 'npm test'
            }
        }
    }
}
